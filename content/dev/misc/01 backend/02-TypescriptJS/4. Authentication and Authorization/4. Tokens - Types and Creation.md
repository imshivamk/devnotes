
token - long string, user and backend has. if it matches, user is authenticated

1. without dara
	a. user has one copy, server-db has one. matched.

2. with data
	a. acces token - short lived
	e.g 10 minutes
	b. refresh token - long lived. e.g 10 days


### access token
- e.g JWT
- stateless, not saved on server. only verified.
- if expired/timed-out, send messege to server with refresh token

### refresh token
- if refresh token sent by user matches, access token is renewed.
- request is intercepted, if access tokens are expired, then refresh token is sent

`.env`
```
ACCES_TOKEN_KEY=test
ACESS_TOKEN_EXPIRY=1d

REFRESH_TOKEN_KEY=test
```

`user.models.js`
```js
import jwt from "jsonwebtokens"
import crypto from "crypto"
import bcrypt from "bcrypt"

userSchema.metods.generateAccessToken = function(){
	return jwt.sign(
		{
			_id: this._id,
			email: this.email,
			username: this.username
		},
		process.env.ACCESS_TOKEN_KEY,
		{expiresIn: process.env.ACCESS_TOKEN_EXPIRY}
	)
}

userSchema.metods.generateRefreshToken = function(){
	return jwt.sign(
		{
			_id: this._id,
		},
		process.env.REFRESH_TOKEN_KEY,
		{expiresIn: process.env.REFRESH_TOKEN_EXPIRY}
	)
}

userSchema.methods.generateTempToken = function(){
	const unhashedToken = crypto.randomBytes(20).
	toString("hex");
	
	const hashedToken = crypto.createHash("sha256").
	update(unhashedToken).digest("hex");
	
	const tokenExpiry = Date.now() + (20*60*1000) //20mins
	
	return {unhashedToken, hashedToken, tokenExpiry}
}

export const User = mongoose.model
```
